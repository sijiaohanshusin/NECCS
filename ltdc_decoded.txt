static uint8_t s_ltdc_dwt_ready = 0u;

static void ltdc_dwt_init(void)
{
    if (s_ltdc_dwt_ready != 0u)
    {
        return;
    }

    CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
    DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;
    DWT->CYCCNT = 0u;
    s_ltdc_dwt_ready = 1u;
}

static void ltdc_delay_us(uint32_t us)
{
    uint32_t ticks;
    uint32_t start;

    if (us == 0u)
    {
        return;
    }

    ltdc_dwt_init();
    ticks = (SystemCoreClock / 1000000u) * us;
    start = DWT->CYCCNT;
    while ((uint32_t)(DWT->CYCCNT - start) < ticks) { }
}

static void ltdc_delay_ms(uint32_t ms)
{
    /* Keep LTDC reset delay independent from RTOS tick to avoid init stalls. */
    while (ms-- > 0u)
    {
        ltdc_delay_us(1000u);
    }
}

#define delay_ms ltdc_delay_ms
#define delay_us ltdc_delay_us

uint32_t *g_ltdc_framebuf[2];              /* LTDC LCD閻㈩垎鍛閻庢稒蓱閺嗙喓绱掗崟顒€鐦归梺?闊洤鎳橀妴蹇涘箰閸パ勫€婚悗鐢垫嚀缁ㄥ弶寰勮閻剟鎯冮崟顐㈡暥閻庢稒锚鐏忣垶宕?*/

volatile uint32_t g_ltdc_init_stage = 0u;
volatile uint32_t g_ltdc_dma2d_timeout_count = 0u;
volatile uint16_t g_ltdc_panel_id = 0u;

#define LTDC_DMA2D_TIMEOUT_LOOP   0X1FFFFFU

static void ltdc_fill_sw_rect(uint32_t psx, uint32_t psy, uint32_t pex, uint32_t pey, uint32_t color)
{
#if LTDC_PIXFORMAT == LTDC_PIXFORMAT_ARGB8888
    uint32_t *base = (uint32_t *)g_ltdc_framebuf[lcdltdc.activelayer];
    for (uint32_t y = psy; y <= pey; y++)
    {
        uint32_t *row = base + y * lcdltdc.pwidth + psx;
        for (uint32_t x = psx; x <= pex; x++)
        {
            *row++ = color;
        }
    }
#elif LTDC_PIXFORMAT == LTDC_PIXFORMAT_RGB888
    uint8_t r = (uint8_t)((color >> 16) & 0xFFu);
    uint8_t g = (uint8_t)((color >> 8) & 0xFFu);
    uint8_t b = (uint8_t)(color & 0xFFu);
    uint8_t *base = (uint8_t *)g_ltdc_framebuf[lcdltdc.activelayer];
    for (uint32_t y = psy; y <= pey; y++)
    {
        uint8_t *row = base + (y * lcdltdc.pwidth + psx) * 3u;
        for (uint32_t x = psx; x <= pex; x++)
        {
            row[0] = b;
            row[1] = g;
            row[2] = r;
            row += 3;
        }
    }
#else
    uint16_t c565 = (uint16_t)color;
    uint16_t *base = (uint16_t *)g_ltdc_framebuf[lcdltdc.activelayer];
    for (uint32_t y = psy; y <= pey; y++)
    {
        uint16_t *row = base + y * lcdltdc.pwidth + psx;
        for (uint32_t x = psx; x <= pex; x++)
        {
            *row++ = c565;
        }
    }
#endif
}

static void ltdc_color_fill_sw_rect(uint32_t psx, uint32_t psy, uint32_t pex, uint32_t pey, const uint16_t *color)
{
#if LTDC_PIXFORMAT == LTDC_PIXFORMAT_RGB565
    uint32_t width = pex - psx + 1u;
    uint16_t *base = (uint16_t *)g_ltdc_framebuf[lcdltdc.activelayer];

    for (uint32_t y = psy; y <= pey; y++)
    {
        uint16_t *row = base + y * lcdltdc.pwidth + psx;
        for (uint32_t x = 0u; x < width; x++)
        {
            row[x] = *color++;
        }
    }
#else
    (void)psx;
    (void)psy;
    (void)pex;
    (void)pey;
    (void)color;
#endif
}

static uint8_t ltdc_dma2d_wait_complete(void)
{
    uint32_t timeout = 0u;

    while (1)
    {
        uint32_t isr = DMA2D->ISR;

        if ((isr & DMA2D_FLAG_TC) != 0u)
        {
            DMA2D->IFCR = DMA2D_FLAG_TC;
            return 0u;
        }

        if ((isr & (DMA2D_FLAG_TE | DMA2D_FLAG_CE | DMA2D_FLAG_CAE)) != 0u)
        {
            g_ltdc_dma2d_timeout_count++;
            DMA2D->IFCR = DMA2D_FLAG_TC | DMA2D_FLAG_TE | DMA2D_FLAG_CE | DMA2D_FLAG_CAE | DMA2D_FLAG_CTC | DMA2D_FLAG_TW;
            return 1u;
        }

        timeout++;
        if (timeout > LTDC_DMA2D_TIMEOUT_LOOP)
        {
            g_ltdc_dma2d_timeout_count++;
            DMA2D->IFCR = DMA2D_FLAG_TC | DMA2D_FLAG_TE | DMA2D_FLAG_CE | DMA2D_FLAG_CAE | DMA2D_FLAG_CTC | DMA2D_FLAG_TW;
            return 1u;
        }
    }
}


/**
 * @brief       LTDC鐎殿喒鍋撻柛?
 * @param       sw   : 1,闁瑰灚鎸哥槐? 0,闁稿繑濞婂Λ?
 * @retval      闁?
 */
void ltdc_switch(uint8_t sw)
{
    if (sw)
    {
        __HAL_LTDC_ENABLE(&g_ltdc_handle);   /* 闁瑰灚鎸哥槐鎱橳DC */
    }
    else
    {
        __HAL_LTDC_DISABLE(&g_ltdc_handle);  /* 闁稿繑濞婂Λ纰橳DC */
    }
}

/**
 * @brief       LTDC鐎殿喒鍋撻柛蹇氭珪鐎垫氨鈧鑹鹃惇?
 * @param       layerx      : 0,缂佹鍏涚粩瀵镐沪? 1,缂佹鍏涚花鈺冧沪?
 * @param       sw          : 1,闁瑰灚鎸哥槐?   0,闁稿繑濞婂Λ?
 * @retval      闁?
 */
void ltdc_layer_switch(uint8_t layerx, uint8_t sw)
{
    if (sw) 
    {
        __HAL_LTDC_LAYER_ENABLE(&g_ltdc_handle, layerx);   /* 鐎殿喒鍋撻柛姘辨－ayerx */
    }
    else
    {
        __HAL_LTDC_LAYER_DISABLE(&g_ltdc_handle, layerx);  /* 闁稿繑濞婂Λ纾峚yerx */
    }

    __HAL_LTDC_RELOAD_CONFIG(&g_ltdc_handle);              /* 缂佹柨顑呭畵鍡涙煂瀹ュ棙鐓€闁告梻濮惧ù鍥煀瀹ュ洨鏋?*/
}

/**
 * @brief       LTDC闁哄嫬澧介妵姘跺棘閻熺増鍊婚悹浣稿⒔閻?
 * @param       dir         : 0,缂佹梹鐗曢惈? 1,婵☆垼浜滈惈?
 * @retval      闁?
 */
void ltdc_display_dir(uint8_t dir)
{
    lcdltdc.dir = dir;     /* 闁哄嫬澧介妵姘跺棘閻熺増鍊?*/

    if (dir == 0)          /* 缂佹梹鐗曢惈?*/
    {
        lcdltdc.width = lcdltdc.pheight;
        lcdltdc.height = lcdltdc.pwidth;
    }
    else if (dir == 1)     /* 婵☆垼浜滈惈?*/
    {
        lcdltdc.width = lcdltdc.pwidth;
        lcdltdc.height = lcdltdc.pheight;
    }
}

/**

{ 
#if LTDC_PIXFORMAT == LTDC_PIXFORMAT_ARGB8888

    }
    else                /* 缂佹梹鐗曢惈?*/
    {
        *(uint32_t *)((uint32_t)g_ltdc_framebuf[lcdltdc.activelayer] + lcdltdc.pixsize * (lcdltdc.pwidth * (lcdltdc.pheight - x - 1) + y)) = color;

#elif LTDC_PIXFORMAT == LTDC_PIXFORMAT_RGB888

    if (lcdltdc.dir)     /* 婵☆垼浜滈惈?*/

        *(uint8_t *)((uint32_t)g_ltdc_framebuf[lcdltdc.activelayer] + lcdltdc.pixsize * (lcdltdc.pwidth * y + x) + 2) = color >> 16;
    }
    else                /* 缂佹梹鐗曢惈?*/
    {
        *(uint16_t *)((uint32_t)g_ltdc_framebuf[lcdltdc.activelayer] + lcdltdc.pixsize * (lcdltdc.pwidth * (lcdltdc.pheight - x - 1) + y)) = color;
        *(uint8_t *)((uint32_t)g_ltdc_framebuf[lcdltdc.activelayer] + lcdltdc.pixsize * (lcdltdc.pwidth * (lcdltdc.pheight - x - 1) + y) + 2) = color >> 16;
    }
    
#else

    if (lcdltdc.dir)    /* 婵☆垼浜滈惈?*/
    {
        *(uint16_t *)((uint32_t)g_ltdc_framebuf[lcdltdc.activelayer] + lcdltdc.pixsize * (lcdltdc.pwidth * y + x)) = color;

/**
 * @brief       LTDC濠靛鍋勯崢鏍儗閳轰浇鍩? DMA2D濠靛鍋勯崢?
 * @note       (sx,sy),(ex,ey):濠靛鍋勯崢鏍儗閳轰浇鍩岄悗鐢殿攰椤宕搁幇顓犲灱,闁告牕鎼悡娆愬緞瑜嶉惃顒佺▔?(ex - sx + 1) * (ey - sy + 1)
 *              婵炲鍔嶉崜?sx,ex,濞戞挸绉烽崗妯诲緞瑜岀花鐞筩ddev.width - 1; sy,ey,濞戞挸绉烽崗妯诲緞瑜岀花鐞筩ddev.height - 1
 * @param       sx,sy       : 閻犙冨槻椤劙宕搁幇顓犲灱
 * @param       ex,ey       : 缂備焦鎸诲顐﹀锤閹邦厾鍨?
 * @param       color       : 濠靛鍋勯崢鏍儍閸曨垼鏉归柤?
 * @retval      闁?
 */
void ltdc_fill(uint16_t sx, uint16_t sy, uint16_t ex, uint16_t ey, uint32_t color)
{ 
    uint32_t psx, psy, pex, pey;        /* 濞寸媭妾糃D闂傚牄鍨哄妯荤▔閸濆嫮鍞ㄩ柛鎴濇濞堟垿宕搁幇顓犲灱缂?濞戞挸绉瑰▓銏犖熼鍡欏建閻忕偛绻愯ぐ澶愬礌閺嶎兘鍋撶仦钘夌秮闁?*/
    uint16_t offline;
    uint32_t addr; 

    if ((lcdltdc.width == 0u) || (lcdltdc.height == 0u))
    {
        return;
    }
    if ((sx >= lcdltdc.width) || (sy >= lcdltdc.height))
    {
        return;
    }
    if (ex >= lcdltdc.width)
    {
        ex = lcdltdc.width - 1u;
    }
    if (ey >= lcdltdc.height)
    {
        ey = lcdltdc.height - 1u;
    }
    if ((ex < sx) || (ey < sy))
    {
        return;
    }

    /* 闁秆勫姈閻栵絿鍖栭弰蹇旂ギ闁?*/
    if (lcdltdc.dir)                    /* 婵☆垼浜滈惈?*/
    {
        psx = sx;
        psy = sy;
        pex = ex;
        pey = ey;
    }
    else                                /* 缂佹梹鐗曢惈?*/
    {
        if (ex >= lcdltdc.pheight)
        {
            ex = lcdltdc.pheight - 1;   /* 闂傚嫭鍔曢崺妤呮嚑閸愩劍绾?*/
        }

        if (sx >= lcdltdc.pheight)
        {
            sx = lcdltdc.pheight - 1;   /* 闂傚嫭鍔曢崺妤呮嚑閸愩劍绾?*/
        }
        
        psx = sy;
        psy = lcdltdc.pheight - ex - 1;
        pex = ey;
        pey = lcdltdc.pheight - sx - 1;
    }

#if (LTDC_ENABLE_DMA2D == 0)
    ltdc_fill_sw_rect(psx, psy, pex, pey, color);
    return;
#endif

    offline = lcdltdc.pwidth - (pex - psx + 1);   /* 閻炴稑鑻禍鍝ョ矓?鐟滅増鎸告晶鐘垫偘鐏炵偓浠橀柛姘凹缁斿瓨绋夐鍕壖缂佽京濮撮幏鐗堢▔鐎ｂ晝顏遍悶娑樼灱椤戝洦绋夐埀顒佺▔椤忓嫬鍓肩紒杈╁С缁狅綁姊荤€靛憡鐣遍柛宥呯箳缁€宀勫极閹殿喗绐?*/
    addr = ((uint32_t)g_ltdc_framebuf[lcdltdc.activelayer] + lcdltdc.pixsize * (lcdltdc.pwidth * psy + psx));

    __HAL_RCC_DMA2D_CLK_ENABLE();                             /* 濞达綀鍎婚崗妤2D闁哄啫鐖奸幐?*/

    DMA2D->CR &= ~(DMA2D_CR_START);                           /* 闁稿繐鐗嗘禒鐘差潰椤㈢嫻A2D */
    DMA2D->IFCR = DMA2D_FLAG_TC | DMA2D_FLAG_TE | DMA2D_FLAG_CE | DMA2D_FLAG_CAE | DMA2D_FLAG_CTC | DMA2D_FLAG_TW;
    DMA2D->CR = DMA2D_R2M;                                    /* 閻庨潧瀚悺銊╁闯閵娿儱鐓傞悗娑櫭崑宥夊闯閵娢翠礁顕?*/
    DMA2D->OPFCCR = LTDC_PIXFORMAT;                           /* 閻犱礁澧介悿鍡橈紣濠婂棗顥忛柡宥囧帶缁?*/
    DMA2D->OOR = offline;                                     /* 閻犱礁澧介悿鍡欐偘鐏炴垝鐒婄紒? */

    DMA2D->OMAR = addr;                                       /* 閺夊牊鎸搁崵顓犫偓娑櫭崑宥夊闯閵娿儲鍕鹃柛褉鍋?*/
    DMA2D->NLR = (pey - psy + 1) | ((pex - psx + 1) << 16);   /* 閻犱焦鍎抽悾鍓ф偘鐏炵偓娈堕悗闈涘閻°劑宕?*/
    DMA2D->OCOLR = color;                                     /* 閻犱焦鍎抽悾鐐綇閹惧啿姣夊Λ鐗堢矎婢瑰﹦鈧潧瀚悺銊╁闯?*/
    DMA2D->CR |= DMA2D_CR_START;                              /* 闁告凹鍨版慨銆A2D */
    if (ltdc_dma2d_wait_complete() != 0u)
    {
        /* DMA2D error/timeout: force software path so init can still paint framebuffer. */
        ltdc_fill_sw_rect(psx, psy, pex, pey, color);
    }
}

///**

/**
 * @brief       闁革负鍔嶇€垫氨鈧鑹剧亸顖炲春閻斿嘲鏁跺┑澶樺亜閸樻牠骞愰崶褏鏆板Λ鐗堢矎婢瑰﹪宕? DMA2D濠靛鍋勯崢?
 * @note        婵縿鍊曢崵閬嶅极妫颁胶鐭岄柡鈧娑樼槷uint16_t,RGB565闁哄秶鍘х槐锟犳儍閸曨垼鏉归柤鐟板级閺嗙喓绱掗崟顐稏闁?
 *              (sx,sy),(ex,ey):濠靛鍋勯崢鏍儗閳轰浇鍩岄悗鐢殿攰椤宕搁幇顓犲灱,闁告牕鎼悡娆愬緞瑜嶉惃顒佺▔?(ex - sx + 1) * (ey - sy + 1)
 *              婵炲鍔嶉崜?sx,ex,濞戞挸绉烽崗妯诲緞瑜岀花鐞筩ddev.width - 1; sy,ey,濞戞挸绉烽崗妯诲緞瑜岀花鐞筩ddev.height - 1
 * @param       sx,sy       : 閻犙冨槻椤劙宕搁幇顓犲灱
 * @param       ex,ey       : 缂備焦鎸诲顐﹀锤閹邦厾鍨?
 * @param       color       : 濠靛鍋勯崢鏍儍閸曨垼鏉归柤鐟板级閺嗙喓绱掗崟顖ｆ禃闁革附婢樺?
 * @retval      闁?
 */
void ltdc_color_fill(uint16_t sx, uint16_t sy, uint16_t ex, uint16_t ey, uint16_t *color)
{
    uint32_t psx, psy, pex, pey;   /* 濞寸媭妾糃D闂傚牄鍨哄妯荤▔閸濆嫮鍞ㄩ柛鎴濇濞堟垿宕搁幇顓犲灱缂?濞戞挸绉瑰▓銏犖熼鍡欏建閻忕偛绻愯ぐ澶愬礌閺嶎兘鍋撶仦钘夌秮闁?*/
    uint16_t offline;
    uint32_t addr;

    if ((lcdltdc.width == 0u) || (lcdltdc.height == 0u) || (color == NULL))
    {
        return;
    }
    if ((sx >= lcdltdc.width) || (sy >= lcdltdc.height))
    {
        return;
    }
    if (ex >= lcdltdc.width)
    {
        ex = lcdltdc.width - 1u;
    }
    if (ey >= lcdltdc.height)
    {
        ey = lcdltdc.height - 1u;
    }
    if ((ex < sx) || (ey < sy))
    {
        return;
    }
  
    /* 闁秆勫姈閻栵絿鍖栭弰蹇旂ギ闁?*/
    if (lcdltdc.dir)               /* 婵☆垼浜滈惈?*/
    {
        psx = sx;
        psy = sy;
        pex = ex;
        pey = ey;
    }
    else                           /* 缂佹梹鐗曢惈?*/
    {
        psx = sy;
        psy = lcdltdc.pheight - ex - 1;
        pex = ey;
        pey = lcdltdc.pheight - sx - 1;
    }

#if (LTDC_ENABLE_DMA2D == 0)
    ltdc_color_fill_sw_rect(psx, psy, pex, pey, color);
    return;
#endif

    offline = lcdltdc.pwidth - (pex - psx + 1);   /* 閻炴稑鑻禍鍝ョ矓?鐟滅増鎸告晶鐘垫偘鐏炵偓浠橀柛姘凹缁斿瓨绋夐鍕壖缂佽京濮撮幏鐗堢▔鐎ｂ晝顏遍悶娑樼灱椤戝洦绋夐埀顒佺▔椤忓嫬鍓肩紒杈╁С缁狅綁姊荤€靛憡鐣遍柛宥呯箳缁€宀勫极閹殿喗绐?*/
    addr = ((uint32_t)g_ltdc_framebuf[lcdltdc.activelayer] + lcdltdc.pixsize * (lcdltdc.pwidth * psy + psx));

    __HAL_RCC_DMA2D_CLK_ENABLE();                             /* 濞达綀鍎婚崗妤2D闁哄啫鐖奸幐?*/

    DMA2D->CR &= ~(DMA2D_CR_START);                           /* 闁稿繐鐗嗘禒鐘差潰椤㈢嫻A2D */
    DMA2D->IFCR = DMA2D_FLAG_TC | DMA2D_FLAG_TE | DMA2D_FLAG_CE | DMA2D_FLAG_CAE | DMA2D_FLAG_CTC | DMA2D_FLAG_TW;
    DMA2D->CR = DMA2D_M2M;                                    /* 閻庢稒锚閸嬪秹宕抽妸銉ョ厒閻庢稒锚閸嬪秹宕抽妸未浣割嚕?*/
    DMA2D->FGPFCCR = LTDC_PIXFORMAT;                          /* 閻犱礁澧介悿鍡涘礈瀹ュ棙鐝悘鐐插€块·渚€鎳濋崣澶屽鐎?*/
    DMA2D->FGOR = 0;                                          /* 闁告挸绉靛▍娆戜沪閸屾繍鏀介柛瀣箳浜涘☉? */
    DMA2D->OOR = offline;                                     /* 閻犱礁澧介悿鍡欐偘鐏炴垝鐒婄紒?*/

    DMA2D->FGMAR = (uint32_t)color;                           /* 婵犙勫姇濠€鎾锤閳?*/
    DMA2D->OMAR = addr;                                       /* 閺夊牊鎸搁崵顓犫偓娑櫭崑宥夊闯閵娿儲鍕鹃柛褉鍋?*/
    DMA2D->NLR = (pey - psy + 1) | ((pex - psx + 1) << 16);   /* 閻犱焦鍎抽悾鍓ф偘鐏炵偓娈堕悗闈涘閻°劑宕?*/
    DMA2D->CR |= DMA2D_CR_START;                              /* 闁告凹鍨版慨銆A2D */
    if (ltdc_dma2d_wait_complete() != 0u)
    {
        /* DMA2D error/timeout: force software path so init can still paint framebuffer. */
        ltdc_color_fill_sw_rect(psx, psy, pex, pey, color);
    }
}  

/**
 * @brief       LTDC婵炴挸鎳庨惈?
 * @param       color       : 濡増绮忔竟濠囧磹?
 * @retval      闁?
 */
void ltdc_clear(uint32_t color)
{
    if ((lcdltdc.width == 0u) || (lcdltdc.height == 0u))
    {
        return;
    }
    ltdc_fill(0, 0, lcdltdc.width - 1, lcdltdc.height - 1, color);
}

    g_ltdc_handle.Init.Backcolor.Blue = 0;                                                          /* 闁煎啿鏈▍娆撴嚌閼艰埖鎲柤瑙勫絻閳?*/
    g_ltdc_init_stage = 4u;
    HAL_LTDC_Init(&g_ltdc_handle);
    g_ltdc_init_stage = 5u;

    /* 閻忕偛鍊块崢銈囩磾?*/
    ltdc_layer_parameter_config(0, (uint32_t)g_ltdc_framebuf[0], LTDC_PIXFORMAT, 255, 0, 6, 7, 0X000000);   /* 閻忕偛鍊稿顒勫极娴兼潙甯崇紓?*/
    ltdc_layer_window_config(0, 0, 0, lcdltdc.pwidth, lcdltdc.pheight);                                     /* 閻忕偛鍊婚悰銉╁矗閿濆甯崇紓?濞寸媭妾糃D闂傚牄鍨哄姗€宕搁幇顓犲灱缂侇垵顔婄拹鐔煎春閸濆嫬娅?濞戞挸绉烽々锕傛⒕韫囧海鈹掑ǎ鍥跺枟閺? */

    //ltdc_layer_parameter_config(1, (uint32_t)g_ltdc_framebuf[1], LTDC_PIXFORMAT, 127, 0, 6, 7, 0X000000); /* 閻忕偛鍊稿顒勫极娴兼潙甯崇紓?*/

    ltdc_select_layer(0);                  /* 闂侇偄顦扮€氥劎绮?閻?*/
    /* Force a first-frame software clear so old SDRAM image is overwritten even if DMA2D is unstable. */
    ltdc_fill_sw_rect(0u, 0u, lcdltdc.pwidth - 1u, lcdltdc.pheight - 1u, 0u);

    /* Turn on backlight early for visibility, even if panel reset sequence stalls. */
    LTDC_BL(1);
    g_ltdc_init_stage = 61u;
    if (lcdid != 0X5571)                   /* 5571鐎规瓕灏欑划鈩冨緞瀹ュ嫮绉撮弶鈺佹矗缁?*/

        LTDC_RST(0);
        g_ltdc_init_stage = 64u;

    g_ltdc_init_stage = 68u;
    LTDC_BL(1);                            /* 闁绘劙鈧稑鐦ㄩ柤鍐茶嫰閸?*/
    g_ltdc_init_stage = 69u;
    ltdc_clear(0XFFFFFFFF);                /* 婵炴挸鎳庨惈?*/
    g_ltdc_init_stage = 70u;
}
